// ARCHIVED: smart scaling (aspect-keeping swap chain), removed 2026-11-27.
// It asked DXGI for DXGI_SCALING_ASPECT_RATIO_STRETCH through the factory's CreateSwapChainForHwnd. The request succeeded
// (result 0), but VIctory makes the swap chain window-sized and scales the picture itself in its own copy, so the flag
// changed nothing visible. A real fix is a fitted destination rectangle in VIctory's presenter copy.

// ---- smart scaling: the factory's CreateSwapChainForHwnd asks for an aspect-keeping swap chain ----
type CreateSwap = unsafe extern "system" fn(*mut c_void, *mut c_void, Hwnd, *const [u32; 12], *const c_void, *mut c_void, *mut *mut c_void) -> i32;
static ORIG_SWAP: std::sync::atomic::AtomicUsize = std::sync::atomic::AtomicUsize::new(0);
static SMART: std::sync::atomic::AtomicBool = std::sync::atomic::AtomicBool::new(false);
unsafe extern "system" fn swap_hook(this: *mut c_void, dev: *mut c_void, hwnd: Hwnd, desc: *const [u32; 12], fs: *const c_void, restrict: *mut c_void, out: *mut *mut c_void) -> i32 {
    let orig: CreateSwap = unsafe { std::mem::transmute(ORIG_SWAP.load(std::sync::atomic::Ordering::Relaxed)) };
    if !desc.is_null() {
        let mut d = unsafe { *desc };
        d[8] = 2; // DXGI_SCALING_ASPECT_RATIO_STRETCH
        let r = unsafe { orig(this, dev, hwnd, &d, fs, restrict, out) };
        eprintln!("uw: swap chain {}x{} asked for aspect-keeping scaling, result {r:#x}", d[0], d[1]);
        if r >= 0 { return r; }
    }
    unsafe { orig(this, dev, hwnd, desc, fs, restrict, out) }
}
fn hook_factory(factory: *mut c_void) {
    use std::sync::atomic::Ordering::Relaxed;
    if factory.is_null() || !SMART.load(Relaxed) || ORIG_SWAP.load(Relaxed) != 0 { return; }
    unsafe {
        let vt = *(factory as *mut *mut usize);
        let slot = vt.add(15);
        let mut old = 0u32;
        if VirtualProtect(slot.cast(), 8, 0x40, &mut old) == 0 { return; }
        ORIG_SWAP.store(*slot, Relaxed);
        *slot = swap_hook as usize;
        VirtualProtect(slot.cast(), 8, old, &mut old);
    }
}
